2. 📝 Struktur Program Arduino

/*
 * Program pertama: LED Blink
 * Komentar multi-baris
 */

// Komentar satu baris

void setup() {
  // Dijalankan sekali di awal
  pinMode(13, OUTPUT);    // Set pin 13 sebagai OUTPUT
  Serial.begin(9600);     // Mulai komunikasi serial
}

void loop() {
  // Dijalankan berulang terus
  digitalWrite(13, HIGH); // Nyalakan LED
  delay(1000);             // Tunggu 1 detik
  digitalWrite(13, LOW);  // Matikan LED
  delay(1000);             // Tunggu 1 detik
}
💡 Penjelasan:
  • setup() : Dijalankan sekali (inisialisasi)
  • loop() : Dijalankan berulang (program utama)
  • pinMode() : Mengatur mode pin (INPUT/OUTPUT)
  • digitalWrite() : Menulis HIGH/LOW ke pin
  • delay() : Menunda program (milidetik)

5. 🔀 Struktur Kontrol

A. if-else

int nilai = 85;

if (nilai >= 60) {
  Serial.println("Lulus");
} else {
  Serial.println("Tidak Lulus");
}

// if-else if
if (nilai >= 85) {
  grade = 'A';
} else if (nilai >= 70) {
  grade = 'B';
} else {
  grade = 'C';
}

B. switch-case

int hari = 3;
String namaHari;

switch (hari) {
  case 1:
    namaHari = "Senin";
    break;
  case 2:
    namaHari = "Selasa";
    break;
  case 3:
    namaHari = "Rabu";
    break;
  default:
    namaHari = "Hari tidak valid";
    break;
}

C. Perulangan (Loop)

// for loop
for (int i = 1; i <= 5; i++) {
  Serial.println(i);
}

// while loop
int i = 1;
while (i <= 5) {
  Serial.println(i);
  i++;
}

// do-while
int i = 1;
do {
  Serial.println(i);
  i++;
} while (i <= 5);

9. 💡 Praktikum 1: LED Blink

A. Rangkaian

Arduino Uno
    pin 13 ────┬───────┐
               │       │
               │    ┌──┴──┐
               │    │ LED │
               └────┴──┬──┘
                     GND

B. Program

const int ledPin = 13;

void setup() {
  pinMode(ledPin, OUTPUT);
  Serial.begin(9600);
}

void loop() {
  digitalWrite(ledPin, HIGH);
  Serial.println("LED ON");
  delay(1000);
  
  digitalWrite(ledPin, LOW);
  Serial.println("LED OFF");
  delay(1000);
}

C. Variasi: LED Fade (PWM)

const int ledPin = 9;  // Pin PWM

void loop() {
  // Fade in
  for (int i = 0; i <= 255; i++) {
    analogWrite(ledPin, i);
    delay(10);
  }
  
  // Fade out
  for (int i = 255; i >= 0; i--) {
    analogWrite(ledPin, i);
    delay(10);
  }
}

10. 🚦 Praktikum 2: Traffic Light

A. Rangkaian

pin 13 ──── Red LED
pin 12 ──── Yellow LED
pin 11 ──── Green LED
GND    ──── Semua LED

B. Program

const int redPin = 13;
const int yellowPin = 12;
const int greenPin = 11;

void setup() {
  pinMode(redPin, OUTPUT);
  pinMode(yellowPin, OUTPUT);
  pinMode(greenPin, OUTPUT);
  Serial.begin(9600);
}

void loop() {
  // Hijau
  digitalWrite(greenPin, HIGH);
  delay(5000);
  digitalWrite(greenPin, LOW);
  
  // Kuning
  digitalWrite(yellowPin, HIGH);
  delay(2000);
  digitalWrite(yellowPin, LOW);
  
  // Merah
  digitalWrite(redPin, HIGH);
  delay(5000);
  digitalWrite(redPin, LOW);
}
